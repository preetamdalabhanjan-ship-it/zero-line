Use Google Drive as the database

1. Open your 5 TB Google Drive → Click New → More → Google Apps Script.

2. Paste the 15-line bridge script below, click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone), and paste the generated URL above:

function doGet(e){var f=DriveApp.getRootFolder().getFilesByName("zeroline_db.json");return ContentService.createTextOutput(f.hasNext()?f.next().getBlob().getDataAsString():"{}").setMimeType(ContentService.MimeType.JSON);}
function doPost(e){var f=DriveApp.getRootFolder().getFilesByName("zeroline_db.json");if(f.hasNext()){f.next().setContent(e.postData.contents);}else{DriveApp.getRootFolder().createFile("zeroline_db.json",e.postData.contents);}return ContentService.createTextOutput('{"ok":true}').setMimeType(ContentService.MimeType.JSON);}
